import kotlin.math.floor

@Serializable
data class SensorTake(
    val hz: Int,
    val kinds: List<String>,
    val frames: List<List<Float>>
) {
    /**
     * The duration of the sensor take in milliseconds.
     * Returns 0 when there are fewer than two frames.
     */
    val durationMs: Long = if (frames.size < 2) 0L else (frames.size - 1) * 1000L / hz
    
    /**
     * Returns the interpolated values at time [tMs].
     * If [loop] is false, times before 0 return the first frame,
     * and times at or past durationMs return the last frame.
     * If [loop] is true, [tMs] is wrapped into the range [0, durationMs).
     */
    fun valueAt(tMs: Long, loop: Boolean = false): List<Float> {
        if (frames.isEmpty()) return emptyList()
        if (frames.size == 1) return frames[0]
        
        val duration = durationMs
        val pos = tMs * hz.toDouble() / 1000.0
        
        if (!loop) {
            if (pos < 0) return frames[0]
            if (pos >= duration) return frames[frames.size - 1]
        } else {
            val wrapped = Math.floorMod(tMs, duration)
            val wrappedPos = wrapped * hz.toDouble() / 1000.0
            val i = floor(wrappedPos).toInt()
            val f = wrappedPos - i
            return interpolate(frames[i], frames[i + 1], f)
        }
        
        val i = floor(pos).toInt()
        val f = pos - i
        return interpolate(frames[i], frames[i + 1], f)
    }
    
    private fun interpolate(frame0: List<Float>, frame1: List<Float>, fraction: Double): List<Float> {
        return frame0.mapIndexed { index, value ->
            value * (1 - fraction) + frame1[index] * fraction
        }
    }
}
