package dk.cbavf.studio.ui.history

class History<T>(initial: T, private val limit: Int = 100,
                 private val clock: () -> Long = { System.currentTimeMillis() }) {
    data class Entry<T>(val label: String, val timeMs: Long, val state: T)
    
    private var entries: MutableList<Entry<T>> = mutableListOf(Entry("Start", clock(), initial))
    private var index: Int = 0
    
    /**
     * Returns a read-only copy of all entries in chronological order.
     */
    val entries: List<Entry<T>>
        get() = entries.toList()
    
    /**
     * Returns the current index position in the history.
     */
    val index: Int
        get() = index
    
    /**
     * Returns the current state.
     */
    val current: T
        get() = entries[index].state
    
    /**
     * Returns true if undo is possible.
     */
    val canUndo: Boolean
        get() = index > 0
    
    /**
     * Returns true if redo is possible.
     */
    val canRedo: Boolean
        get() = index < entries.size - 1
    
    /**
     * Adds a new entry to the history with the given label and state.
     */
    fun push(label: String, state: T) {
        // Drop entries after current index
        if (index < entries.size - 1) {
            entries.subList(index + 1, entries.size).clear()
        }
        
        // Add new entry
        entries.add(Entry(label, clock(), state))
        index = entries.size - 1
        
        // Trim entries if needed
        if (entries.size > limit) {
            val removeCount = entries.size - limit
            entries.subList(0, removeCount).clear()
            index -= removeCount
        }
    }
    
    /**
     * Moves the history back by one entry and returns the new current state.
     * Returns null if undo is not possible.
     */
    fun undo(): T? {
        if (index <= 0) return null
        index--
        return entries[index].state
    }
    
    /**
     * Moves the history forward by one entry and returns the new current state.
     * Returns null if redo is not possible.
     */
    fun redo(): T? {
        if (index >= entries.size - 1) return null
        index++
        return entries[index].state
    }
    
    /**
     * Jumps to a specific entry by index and returns its state.
     * Returns null if the index is invalid.
     */
    fun jumpTo(i: Int): T? {
        if (i < 0 || i >= entries.size) return null
        index = i
        return entries[index].state
    }
    
    /**
     * Returns a list of labels for all entries in chronological order.
     */
    fun labels(): List<String> {
        return entries.map { it.label }
    }
}
